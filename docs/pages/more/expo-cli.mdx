import { useState } from "react";
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";

const missions = [
  {
    id: 1,
    title: "Despertar Consciente",
    time: "07:00",
    description:
      "Acordar sem celular, banho frio ou morno e respiração Box Breathing por 4 minutos."
  },
  {
    id: 2,
    title: "Afirmação Shinobi",
    time: "07:10",
    description:
      "Dizer: 'Hoje eu desperto. Estou fora da ilusão. Minha mente me pertence. Meu corpo é minha armadura. Minha alma é fogo eterno. E meu tempo é sagrado.'"
  },
  {
    id: 3,
    title: "Treinamento Corporal",
    time: "07:30",
    description:
      "20 flexões, 20 abdominais, 20 agachamentos e 10 minutos de alongamento."
  },
  {
    id: 4,
    title: "Ritual Mental",
    time: "14:00",
    description:
      "1 hora sem distrações. Refletir: Quem eu seria se tivesse nascido sem medo de nada?"
  },
  {
    id: 5,
    title: "Diário da Ascensão",
    time: "22:00",
    description:
      "Escrever: O que eu controlei hoje? O que me controlou? O que posso melhorar amanhã?"
  }
];

export default function AscensaoShinobi() {
  const [completed, setCompleted] = useState([]);
  const [streak, setStreak] = useState(0);

  const toggleMission = (id) => {
    const updated = completed.includes(id)
      ? completed.filter((i) => i !== id)
      : [...completed, id];
    setCompleted(updated);

    if (updated.length === missions.length) {
      setStreak((prev) => prev + 1);
    }
  };

  return (
    <div className="p-6 space-y-6 min-h-screen bg-gradient-to-br from-black via-gray-900 to-zinc-800 text-white">
      <h1 className="text-3xl font-bold text-center">Jornada de Ascensão Shinobi</h1>
      <p className="text-center text-gray-400">
        Missões Diárias - Semana 1: Despertar do Caminho
      </p>
      <div className="text-center text-yellow-400 font-mono">
        🔥 Dias consecutivos cumprindo todas as missões: {streak}
      </div>
      <div className="grid md:grid-cols-2 gap-4">
        {missions.map((mission) => (
          <Card
            key={mission.id}
            className={
              completed.includes(mission.id)
                ? "border-green-500 bg-green-900"
                : "bg-zinc-900"
            }
          >
            <CardContent className="p-4 space-y-2">
              <div className="flex justify-between items-center">
                <h2 className="text-xl font-semibold text-white">{mission.title}</h2>
                <span className="text-sm text-gray-400">{mission.time}</span>
              </div>
              <p className="text-gray-300">{mission.description}</p>
              <Button
                onClick={() => toggleMission(mission.id)}
                variant={completed.includes(mission.id) ? "outline" : "default"}
              >
                {completed.includes(mission.id) ? "Desmarcar" : "Concluir"}
              </Button>
            </CardContent>
          </Card>
        ))}
      </div>
    </div>
  );
}
